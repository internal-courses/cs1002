    - `C` (Clubs) - 4

    - `A` (Ace) - 1
    - `2` through `10` - same as the number
    - `J` (Jack) - 11
    - `Q` (Queen) - 12
    - `K` (King) - 13

    Examples:
    >>> card_to_value_tuple('AH')
    (2, 1)
    >>> card_to_value_tuple('7D')
    (3, 7)
    >>> card_to_value_tuple('QS')
    (1, 12)
    >>> card_to_value_tuple('9C')
    (4, 9)
    >>> card_to_value_tuple('10D')
    (3, 10)

    Args:
        card1 (str): A valid card string (e.g., "AH", "7D")
        card2 (str): Another valid card string

    Returns:
        str: The higher-value card
    '''
    suit_map = {'S':1,'H':2,'D':3,'C':4}

    suit = card[-1]

    n0=s[0]

    if order==(0, 1, 2):

    elif order==(0, 2, 1):
        return n0+" "+n2+" "+n1
    elif order==(1, 0, 2):
        return n1+" "+n0+" "+n2
    elif order==(1, 2, 0):
        return n1+" "+n2+" "+n0
    elif order==(2, 0, 1):

    else:
